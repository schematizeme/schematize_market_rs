CMD schematize-market
  SOBRE schematize market — install and remove programs
CMD schematize-market desktop
  SOBRE Icon and application-menu entry — so the app opens without the hub
  ARG install              --install
  ARG remove               --remove
CMD schematize-market install
  SOBRE Install a language runtime, a dev tool, or an ecosystem app
  ARG dry_run              --dry-run
  ARG method               --method
  ARG what                  OBRIGATORIO
  ARG yes                  --yes -y
CMD schematize-market list
  SOBRE Everything installable, and what is already installed. This is what the icon opens
  ARG json                 --json
  ARG wait                 --wait
CMD schematize-market pin
  SOBRE Pin the app to a version, so `update` stops moving it. `latest` unpins
  ARG version               OBRIGATORIO
CMD schematize-market remove
  SOBRE Remove a language runtime or a dev tool (auto-detects how it was installed)
  ARG dry_run              --dry-run
  ARG method               --method
  ARG what                  OBRIGATORIO
CMD schematize-market run
  SOBRE Launch the installed GUI by absolute path (does not depend on the desktop PATH)
CMD schematize-market status
  SOBRE Versions, platform, pin, and which components are installed
  ARG json                 --json
CMD schematize-market switch
  SOBRE Switch how a language is installed. Installs the new method FIRST, then removes the old
  ARG dry_run              --dry-run
  ARG lang                  OBRIGATORIO
  ARG to                   --to OBRIGATORIO
  ARG yes                  --yes -y
CMD schematize-market unpin
  SOBRE Stop following a pinned version — same as `pin latest`
CMD schematize-market update
  SOBRE Update everything installed from the ecosystem (app, GUI, apps, and this program)
  ARG dry_run              --dry-run
  ARG force                --force
